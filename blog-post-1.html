Practical SQL Patterns for Analytics

Jan 2025 • 5 min read

Window functions

Rank customers by spend and compute rolling totals.

-- Monthly spend per customer with rank
WITH monthly AS (
  SELECT customer_id, date_trunc('month', order_date) AS month, SUM(amount) AS spend
  FROM orders
  GROUP BY 1,2
)
SELECT
  customer_id,
  month,
  spend,
  SUM(spend) OVER (PARTITION BY customer_id ORDER BY month ROWS BETWEEN 2 PRECEDING AND CURRENT ROW) AS rolling_3m,
  RANK() OVER (PARTITION BY month ORDER BY spend DESC) AS spend_rank
FROM monthly;

Cohort analysis

WITH first_order AS (
  SELECT customer_id, MIN(order_date) AS cohort
  FROM orders
  GROUP BY 1
), orders_labeled AS (
  SELECT o.customer_id, o.order_date, DATE_TRUNC('month', o.order_date) AS month,
         DATE_TRUNC('month', fo.cohort) AS cohort_month
  FROM orders o JOIN first_order fo USING (customer_id)
)
SELECT cohort_month,
       DATE_PART('month', month - cohort_month) AS month_number,
       COUNT(DISTINCT customer_id) AS active_customers
FROM orders_labeled
GROUP BY 1,2
ORDER BY 1,2;
